## Critical Findings from Reconnaissance

Before reading the plan, understand these discoveries that shape every sprint:

### Finding 1: All Core Code is Stubs

Every handler, service, and route returns hardcoded strings or uses in-memory arrays:
- **Connector handlers**: `return { content: [{ type: 'text', text: 'Created issue: ${title}' }] }` — no real GitHub/Jira/Slack API calls
- **Memory**: `const conversations: ConversationRow[] = []` — arrays, not SQLite
- **Orchestrator**: 13 lines, calls planner then iterates `toolCalls` — no real agent loop with maxSteps, tool result feeding
- **Chat route**: Imports manifests directly, not through ConnectorRegistry

### Finding 2: migrate.ts is Incomplete

`apps/server/src/db/migrate.ts` (6 lines) only creates `conversations` and `messages` tables. The `schema.ts` file defines 7 tables: `conversations`, `messages`, `messageAttachments`, `connectorConfigs`, `flowTemplates`, `executionLog`, `modelConfig`. Five tables are never created.

### Finding 3: Frontend Shows Splash Page Only

`apps/web/src/main.tsx` renders a static splash page with "Automate / Freedom is the right of all sentient beings." It does NOT mount any router or render `ChatShell`. The router exists (`router.tsx`) but defines routes as a plain data array — not wired to TanStack Router. The `ChatShell` component (190 lines) is fully built but unreachable.

### Finding 4: Vault is Fully Functional

The vault subsystem (crypto, repository, session, service) is the ONE piece that works end-to-end. AES-256-GCM encryption, PBKDF2 key derivation, SQLite-backed storage. Do not modify.

### Finding 5: Shared Package Uses Zod v3, Connectors Use Zod v4

`packages/shared/src/index.ts` imports from `zod` (v3), while `packages/connector-sdk/src/types.ts` imports from `zod/v4`. This mismatch needs careful handling during integration to prevent runtime schema errors.

### Finding 6: ConnectorRegistry Not Used

`apps/server/src/connectors/registry.ts` exists and works (37 lines with registerLocal/dispatch/unload/reload), but `apps/server/src/routes/chat.ts` bypasses it entirely — importing manifests directly from connector packages. The registry needs to be wired in as the single source of truth.

```ts
import type Database from 'better-sqlite3';
import { db as defaultDb } from './client.js';

export function migrateDb(database?: ReturnType<typeof Database>) {
  const target = database ?? defaultDb;

  target.exec(`
    CREATE TABLE IF NOT EXISTS conversations (
      id TEXT PRIMARY KEY,
      title TEXT,
      created_at TEXT DEFAULT (datetime('now'))
    );

    CREATE TABLE IF NOT EXISTS messages (
      id TEXT PRIMARY KEY,
      conversation_id TEXT NOT NULL REFERENCES conversations(id) ON DELETE CASCADE,
      role TEXT NOT NULL,
      content TEXT NOT NULL,
      tool_invocations TEXT,
      created_at TEXT DEFAULT (datetime('now'))
    );

    CREATE TABLE IF NOT EXISTS message_attachments (
      id TEXT PRIMARY KEY,
      message_id TEXT NOT NULL REFERENCES messages(id) ON DELETE CASCADE,
      type TEXT NOT NULL,
      content TEXT NOT NULL,
      created_at TEXT DEFAULT (datetime('now'))
    );

    CREATE TABLE IF NOT EXISTS connector_configs (
      id TEXT PRIMARY KEY,
      name TEXT NOT NULL UNIQUE,
      display_name TEXT NOT NULL,
      enabled INTEGER NOT NULL DEFAULT 0,
      config TEXT NOT NULL DEFAULT '{}',
      created_at TEXT DEFAULT (datetime('now')),
      updated_at TEXT DEFAULT (datetime('now'))
    );

    CREATE TABLE IF NOT EXISTS flow_templates (
      id TEXT PRIMARY KEY,
      name TEXT NOT NULL UNIQUE,
      description TEXT,
      system_prompt TEXT NOT NULL,
      tool_names TEXT NOT NULL DEFAULT '[]',
      created_at TEXT DEFAULT (datetime('now'))
    );

    CREATE TABLE IF NOT EXISTS execution_log (
      id TEXT PRIMARY KEY,
      conversation_id TEXT NOT NULL REFERENCES conversations(id) ON DELETE CASCADE,
      tool_name TEXT NOT NULL,
      input TEXT NOT NULL,
      output TEXT,
      status TEXT NOT NULL DEFAULT 'running',
      duration_ms INTEGER,
      error_message TEXT,
      created_at TEXT DEFAULT (datetime('now'))
    );

    CREATE TABLE IF NOT EXISTS model_config (
      id TEXT PRIMARY KEY DEFAULT 'default',
      provider TEXT NOT NULL DEFAULT 'ollama',
      model TEXT NOT NULL DEFAULT 'llama3.1',
      endpoint TEXT NOT NULL DEFAULT 'http://localhost:11434',
      temperature REAL DEFAULT 0.7,
      max_tokens INTEGER DEFAULT 4096,
      updated_at TEXT DEFAULT (datetime('now'))
    );
  `);
}

/** Backward-compat default export */
export function migrate() {
  migrateDb();
}
```

```ts
import { eq, desc } from 'drizzle-orm';
import type { BetterSQLite3Database } from 'drizzle-orm/better-sqlite3';
import { conversations, messages } from '../db/schema.js';
import type { MemoryRepository } from './memory.js';

export function createDrizzleMemory(db: BetterSQLite3Database): MemoryRepository {
  return {
    async saveConversation(conv) {
      db.insert(conversations).values({
        id: conv.id,
        title: conv.title,
      }).run();
    },

    async getConversation(id) {
      const row = db.select().from(conversations).where(eq(conversations.id, id)).get();
      return row ?? null;
    },

    async listConversations() {
      return db.select().from(conversations).orderBy(desc(conversations.createdAt)).all();
    },

    async deleteConversation(id) {
      // Messages cascade via FK, but SQLite needs PRAGMA foreign_keys=ON (set in client.ts)
      db.delete(messages).where(eq(messages.conversationId, id)).run();
      db.delete(conversations).where(eq(conversations.id, id)).run();
    },

    async saveMessage(msg) {
      db.insert(messages).values({
        id: msg.id,
        conversationId: msg.conversationId,
        role: msg.role,
        content: msg.content,
        toolInvocations: msg.toolInvocations ?? null,
      }).run();
    },

    async listMessages(conversationId) {
      return db.select().from(messages).where(eq(messages.conversationId, conversationId)).orderBy(messages.createdAt).all();
    },
  };
}
```

```ts
import type { ConnectorManifest } from '@automate/connector-sdk';

export class ConnectorRegistry {
  private connectors = new Map<string, ConnectorManifest>();

  registerManifest(manifest: ConnectorManifest) {
    this.connectors.set(manifest.name, manifest);
  }

  listManifests(): ConnectorManifest[] {
    return [...this.connectors.values()];
  }

  async dispatch(toolName: string, input: unknown, credentials: Record<string, string> = {}) {
    const [connectorName, toolMethodName] = toolName.split('.');
    const manifest = this.connectors.get(connectorName);
    if (!manifest) throw new Error(`Connector not loaded: ${connectorName}`);
    const tool = manifest.tools.find(t => t.name === toolMethodName);
    if (!tool) throw new Error(`Tool not found: ${toolName}`);
    return tool.handler(input, { credentials, abortSignal: new AbortController().signal });
  }

  unload(name: string) {
    this.connectors.delete(name);
  }

  reload(manifest: ConnectorManifest) {
    this.unload(manifest.name);
    this.registerManifest(manifest);
  }
}
```

```ts
import { streamText, tool } from 'ai';
import { createOllama } from 'ollama-ai-provider';
import type { ConnectorRegistry } from '../connectors/registry.js';
import type { ModelConfig } from './planner.js';

export function createOrchestrator(registry: ConnectorRegistry) {
  return {
    buildStreamParams(
      messages: Array<{ role: string; content: string }>,
      systemPrompt: string,
      modelConfig: ModelConfig,
      credentials: Record<string, Record<string, string>>,
    ) {
      const ollama = createOllama({ baseURL: `${modelConfig.endpoint}/api` });
      const manifests = registry.listManifests();

      // Convert connector tools to AI SDK tools
      const tools: Record<string, any> = {};
      for (const manifest of manifests) {
        const connectorCreds = credentials[manifest.name] ?? {};
        for (const t of manifest.tools) {
          const toolKey = `${manifest.name}__${t.name}`;
          tools[toolKey] = tool({
            description: t.description,
            parameters: t.inputSchema,
            execute: async (input) => {
              const result = await registry.dispatch(`${manifest.name}.${t.name}`, input, connectorCreds);
              return result.content.map(c => c.text).join('\n');
            },
          });
        }
      }

      return {
        model: ollama(modelConfig.model),
        system: systemPrompt,
        messages: messages as any,
        tools,
        maxSteps: 5,
        temperature: modelConfig.temperature ?? 0.7,
        maxTokens: modelConfig.maxTokens ?? 4096,
      };
    },

    stream(
      messages: Array<{ role: string; content: string }>,
      systemPrompt: string,
      modelConfig: ModelConfig,
      credentials: Record<string, Record<string, string>>,
    ) {
      const params = this.buildStreamParams(messages, systemPrompt, modelConfig, credentials);
      return streamText(params);
    },
  };
}
```

```ts
import type { FastifyInstance } from 'fastify';
import type { MemoryRepository } from '../agent/memory.js';
import type { ConnectorRegistry } from '../connectors/registry.js';
import { createOrchestrator } from '../agent/orchestrator-loop.js';
import type { ModelConfig } from '../agent/planner.js';

interface ChatDeps {
  memory: MemoryRepository;
  registry: ConnectorRegistry;
  getModelConfig: () => Promise<ModelConfig>;
  getCredentials: () => Promise<Record<string, Record<string, string>>>;
  getSystemPrompt: () => string;
}

export async function chatRoutes(app: FastifyInstance, deps: ChatDeps) {
  const orchestrator = createOrchestrator(deps.registry);

  app.post('/api/chat', async (request, reply) => {
    const body = request.body as {
      messages: Array<{ role: string; content: string }>;
      conversationId?: string;
    };

    // Ensure conversation exists
    let conversationId = body.conversationId;
    if (!conversationId) {
      conversationId = crypto.randomUUID();
      const firstMsg = body.messages.find(m => m.role === 'user');
      await deps.memory.saveConversation({
        id: conversationId,
        title: firstMsg?.content.slice(0, 100) ?? 'New Chat',
      });
    }

    // Save user message
    const userMsg = body.messages[body.messages.length - 1];
    if (userMsg?.role === 'user') {
      await deps.memory.saveMessage({
        id: crypto.randomUUID(),
        conversationId,
        role: 'user',
        content: userMsg.content,
      });
    }

    // Stream response
    const modelConfig = await deps.getModelConfig();
    const credentials = await deps.getCredentials();
    const systemPrompt = deps.getSystemPrompt();

    const result = orchestrator.stream(
      body.messages,
      systemPrompt,
      modelConfig,
      credentials,
    );

    // Use Vercel AI SDK's built-in streaming response
    return (await result).toDataStreamResponse();
  });
}
```

```ts
import { z } from 'zod';
import { Octokit } from '@octokit/rest';
import type { ConnectorManifest, ToolContext } from '@automate/connector-sdk';

function getOctokit(ctx: ToolContext) {
  return new Octokit({ auth: ctx.credentials.token });
}

export const githubManifest: ConnectorManifest = {
  name: 'github',
  version: '0.1.0',
  displayName: 'GitHub',
  description: 'GitHub MCP connector — create issues, comment on PRs',
  icon: 'github',
  credentialSchema: z.object({ token: z.string(), owner: z.string(), repo: z.string() }),
  tools: [
    {
      name: 'create_issue',
      description: 'Create a GitHub issue in the configured repository',
      inputSchema: z.object({ title: z.string(), body: z.string(), labels: z.array(z.string()).optional() }),
      handler: async (input: any, ctx: ToolContext) => {
        const octokit = getOctokit(ctx);
        const { owner, repo } = ctx.credentials;
        const issue = await octokit.issues.create({
          owner, repo,
          title: input.title,
          body: input.body,
          labels: input.labels,
```

## Appendix: Zod v3/v4 Compatibility Note

`packages/shared` uses Zod v3, while `packages/connector-sdk` uses `zod/v4`. During integration:
- Connector schemas (tool inputs, credential schemas) use Zod v4
- Shared package schemas (domain types) use Zod v3
- The Vercel AI SDK's `tool()` function needs Zod v3 schemas
- Resolution: Use `zod` (v3) in tool definitions passed to AI SDK, keep connector-sdk types as `zod/v4` internally
